예시 사진 다운로드 + 분석
import urllib.request from IPython.display import display import PIL.Image # 예시 사진 목록 (pmimage) SAMPLE_IMGS = [ "https://6250hijk-boop.github.io/ai-class-game/images/pmimage/pm1.jpg", "https://6250hijk-boop.github.io/ai-class-game/images/pmimage/pm2.jpg", "https://6250hijk-boop.github.io/ai-class-game/images/pmimage/pm3.jpg", ] print("📸 예시 사진 분석 시작!") print("="*50) for i, url in enumerate(SAMPLE_IMGS): try: # 사진 다운로드 fname = f"sample_{i+1}.jpg" urllib.request.urlretrieve(url, fname) # AI 분석 results = model(fname, conf=0.3) result_img = PIL.Image.fromarray(results[0].plot()[..., ::-1]) display(result_img) # 결과 출력 print(f"\n📋 예시 사진 {i+1} 분석 결과:") if results[0].boxes: for box in results[0].boxes: cls_name = model.names[int(box.cls)] conf = float(box.conf) emoji = "🛴" if "킥보드" in cls_name else "⛑️" if "헬멧" in cls_name else "🧍" print(f" {emoji} {cls_name}: {conf:.1%}") else: print(" 감지된 객체가 없습니다.") print("-"*30) except Exception as e: print(f" ⚠️ 사진 {i+1} 처리 실패: {e}")

STEP 3 📸 내 사진 업로드해서 분석하기

1
사진 1장 업로드 + 분석
from google.colab import files from IPython.display import display import PIL.Image, io print("📤 분석할 사진을 업로드하세요!") uploaded = files.upload() for fname, data in uploaded.items(): print(f"\n🔍 '{fname}' 분석 중...") pil_img = PIL.Image.open(io.BytesIO(data)) results = model(pil_img, conf=0.3) # 결과 이미지 표시 result_img = PIL.Image.fromarray(results[0].plot()[..., ::-1]) display(result_img) # 감지 결과 출력 print(f"\n📋 분석 결과:") if results[0].boxes: counts = {} for box in results[0].boxes: cls_name = model.names[int(box.cls)] conf = float(box.conf) counts[cls_name] = counts.get(cls_name, 0) + 1 emoji = "🛴" if "킥보드" in cls_name else "⛑️" if "헬멧" in cls_name else "🧍" print(f" {emoji} {cls_name}: {conf:.1%}") print(f"\n📊 요약: {dict(counts)}") # 안전모 착용 여부 판단 if "킥보드" in str(counts) or "사람" in str(counts): if "헬멧" in str(counts): print("✅ 헬멧 착용이 감지되었습니다!") else: print("⚠️ 헬멧 미착용이 의심됩니다!") else: print(" 감지된 객체가 없습니다.")
2
여러 장 한번에 분석
from google.colab import files from IPython.display import display import PIL.Image, io print("📤 여러 사진을 한번에 업로드하세요!") uploaded = files.upload() total = len(uploaded) helmet_count = 0 no_helmet_count = 0 print(f"\n총 {total}장 분석 시작!") print("="*50) for i, (fname, data) in enumerate(uploaded.items()): print(f"\n[{i+1}/{total}] {fname}") pil_img = PIL.Image.open(io.BytesIO(data)) results = model(pil_img, conf=0.3) result_img = PIL.Image.fromarray(results[0].plot()[..., ::-1]) display(result_img) classes_found = [model.names[int(b.cls)] for b in results[0].boxes] has_person_or_kickboard = any("킥보드" in c or "사람" in c for c in classes_found) has_helmet = any("헬멧" in c for c in classes_found) if has_person_or_kickboard: if has_helmet: helmet_count += 1 print(" ✅ 헬멧 착용") else: no_helmet_count += 1 print(" ⚠️ 헬멧 미착용") print(f"\n{'='*50}") print(f"📊 최종 결과:") print(f" ✅ 헬멧 착용: {helmet_count}건") print(f" ⚠️ 헬멧 미착용: {no_helmet_count}건")

from google.colab import files import os print("📤 분석할 사진을 업로드하세요!") uploaded = files.upload() saved_files = [] for fname, data in uploaded.items(): pil_img = PIL.Image.open(io.BytesIO(data)) results = model(pil_img, conf=0.3) # 결과 이미지 저장 result_img = PIL.Image.fromarray(results[0].plot()[..., ::-1]) save_name = f"result_{fname}" result_img.save(save_name) saved_files.append(save_name) display(result_img) print(f" ✅ {save_name} 저장됨") # 다운로드 print("\n📥 결과 이미지 다운로드 중...") for f in saved_files: files.download(f) print("✅ 완료!")
