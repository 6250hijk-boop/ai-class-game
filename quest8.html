from ultralytics import YOLO import PIL.Image from IPython.display import display import glob # 학습된 모델 로드 (Drive에서) model = YOLO(f'{DATA_DIR}/runs/my_model/weights/best.pt') # val 폴더의 사진으로 테스트 (학습 안 한 사진!) test_imgs = glob.glob(f'{DATA_DIR}/images/val/*')[:5] # 5장만 if not test_imgs: print("⚠️ val 폴더에 사진이 없어요. train 폴더로 대체합니다.") test_imgs = glob.glob(f'{DATA_DIR}/images/train/*')[:5] results = model(test_imgs, conf=0.3) for r in results: img = PIL.Image.fromarray(r.plot()[..., ::-1]) display(img) print(f"📸 {r.path.split('/')[-1]}") if r.boxes: for box in r.boxes: cls_name = model.names[int(box.cls)] conf = float(box.conf) print(f" → {cls_name}: {conf:.1%}") print()

!pip install ultralytics -q import zipfile, os # ZIP 파일 경로 (본인 Drive 경로로 수정) zip_path = '/content/drive/MyDrive/ai-class/dataset.zip' extract_path = '/content/dataset' # ZIP 압축 해제 with zipfile.ZipFile(zip_path, 'r') as z: z.extractall(extract_path) print("✅ 데이터 준비 완료!") print(os.listdir(extract_path))
3
학습 데이터 train/val 분리
import shutil, random img_dir = '/content/dataset/images' lbl_dir = '/content/dataset/labels' # train/val 폴더 생성 for d in ['images/train','images/val','labels/train','labels/val']: os.makedirs(f'/content/dataset/{d}', exist_ok=True) # 이미지 목록 imgs = [f for f in os.listdir(img_dir) if f.lower().endswith(('.jpg','.jpeg','.png','.webp')) and os.path.isfile(f'{img_dir}/{f}')] random.shuffle(imgs) # 80% train, 20% val split = int(len(imgs) * 0.8) train_imgs = imgs[:split] val_imgs = imgs[split:] for fname in train_imgs: shutil.move(f'{img_dir}/{fname}', f'/content/dataset/images/train/{fname}') base = os.path.splitext(fname)[0] txt = f'{lbl_dir}/{base}.txt' if os.path.exists(txt): shutil.move(txt, f'/content/dataset/labels/train/{base}.txt') for fname in val_imgs: shutil.move(f'{img_dir}/{fname}', f'/content/dataset/images/val/{fname}') base = os.path.splitext(fname)[0] txt = f'{lbl_dir}/{base}.txt' if os.path.exists(txt): shutil.move(txt, f'/content/dataset/labels/val/{base}.txt') print(f"✅ train: {len(train_imgs)}장 / val: {len(val_imgs)}장")
4
YOLOv8 학습 시작!
from ultralytics import YOLO model = YOLO('yolov8n.pt') results = model.train( data='/content/dataset/dataset.yaml', epochs=50, imgsz=640, batch=16, name='my_model', project='/content/drive/MyDrive/ai-class/runs', save=True, ) print("🎉 학습 완료!") print("best.pt 위치:", results.save_dir + '/weights/best.pt')
💡 데이터가 적으면 epochs=30으로 줄이세요. 학습 시간: 약 5~15분
