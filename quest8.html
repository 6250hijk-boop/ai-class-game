STEP 3 🌐 Streamlit으로 테스트하기

⚠️ Streamlit은 별도 서버가 필요합니다. Colab에서 ngrok으로 실행하거나 Streamlit Cloud를 사용하세요.
1
아래 코드를 app.py로 저장하세요:
import streamlit as st from ultralytics import YOLO from PIL import Image import numpy as np import cv2 st.title("🤖 내가 만든 AI 테스트!") st.write("직접 라벨링하고 학습시킨 AI로 사물을 인식해보세요.") model_path = st.text_input("best.pt 경로", "best.pt") conf = st.slider("신뢰도 임계값", 0.1, 1.0, 0.5) uploaded = st.file_uploader("테스트할 사진 업로드", type=["jpg","jpeg","png","webp"]) if uploaded and model_path: model = YOLO(model_path) image = Image.open(uploaded) img_array = np.array(image) results = model(img_array, conf=conf) # 결과 시각화 result_img = results[0].plot() result_pil = Image.fromarray(result_img[...,::-1]) col1, col2 = st.columns(2) with col1: st.write("**원본**") st.image(image, use_container_width=True) with col2: st.write("**인식 결과**") st.image(result_pil, use_container_width=True) # 결과 표 if results[0].boxes: st.write("**감지된 객체:**") for box in results[0].boxes: cls = model.names[int(box.cls)] conf_val = float(box.conf) st.write(f"- {cls}: {conf_val:.1%}")
2
Colab에서 실행하려면:
!pip install streamlit pyngrok -q !ngrok authtoken YOUR_TOKEN # ngrok 토큰 입력 import threading def run(): !streamlit run app.py --server.port 8501 t = threading.Thread(target=run) t.start() from pyngrok import ngrok url = ngrok.connect(8501) print("🌐 접속 URL:", url)
