학습 데이터 train/val 분리
import shutil, random img_dir = '/content/dataset/images' lbl_dir = '/content/dataset/labels' # train/val 폴더 생성 for d in ['images/train','images/val','labels/train','labels/val']: os.makedirs(f'/content/dataset/{d}', exist_ok=True) # 이미지 목록 imgs = [f for f in os.listdir(img_dir) if f.lower().endswith(('.jpg','.jpeg','.png','.webp')) and os.path.isfile(f'{img_dir}/{f}')] random.shuffle(imgs) # 80% train, 20% val split = int(len(imgs) * 0.8) train_imgs = imgs[:split] val_imgs = imgs[split:] for fname in train_imgs: shutil.move(f'{img_dir}/{fname}', f'/content/dataset/images/train/{fname}') base = os.path.splitext(fname)[0] txt = f'{lbl_dir}/{base}.txt' if os.path.exists(txt): shutil.move(txt, f'/content/dataset/labels/train/{base}.txt') for fname in val_imgs: shutil.move(f'{img_dir}/{fname}', f'/content/dataset/images/val/{fname}') base = os.path.splitext(fname)[0] txt = f'{lbl_dir}/{base}.txt' if os.path.exists(txt): shutil.move(txt, f'/content/dataset/labels/val/{base}.txt') print(f"✅ train: {len(train_imgs)}장 / val: {len(val_imgs)}장")
4
YOLOv8 학습 시작!
from ultralytics import YOLO model = YOLO('yolov8n.pt') # nano 모델 (가장 빠름) results = model.train( data='/content/dataset/dataset.yaml', epochs=50, # 학습 횟수 (적을수록 빠름) imgsz=640, # 이미지 크기 batch=16, # 배치 크기 name='my_model', project='/content/drive/MyDrive/ai-class/runs', save=True, ) print("🎉 학습 완료!") print("best.pt 위치:", results.save_dir + '/weights/best.pt')
💡 데이터가 적으면 epochs=30으로 줄이세요. 학습 시간: 약 5~15분

STEP 3 🎯 Colab에서 바로 테스트하기

1
테스트할 사진 업로드 & 결과 확인
from ultralytics import YOLO from google.colab import files from IPython.display import display import PIL.Image, io # 학습된 모델 로드 model = YOLO('/content/drive/MyDrive/ai-class/runs/my_model/weights/best.pt') # 테스트 사진 업로드 print("📸 테스트할 사진을 업로드하세요!") uploaded = files.upload() for fname, data in uploaded.items(): # 예측 results = model(io.BytesIO(data), conf=0.3) # 결과 이미지 출력 result_img = results[0].plot() img = PIL.Image.fromarray(result_img[..., ::-1]) display(img) # 감지된 객체 출력 print(f"\n📋 감지된 객체 ({fname}):") if results[0].boxes: for box in results[0].boxes: cls_name = model.names[int(box.cls)] conf = float(box.conf) print(f" → {cls_name}: {conf:.1%} 신뢰도") else: print(" 감지된 객체가 없습니다.")
2
여러 사진 한번에 테스트
import glob # 테스트 이미지 폴더 지정 test_imgs = glob.glob('/content/dataset/images/val/*')[:5] # val 5장 results = model(test_imgs, conf=0.3) for r in results: img = PIL.Image.fromarray(r.plot()[..., ::-1]) display(img) print(f"파일: {r.path}")
💡 val 폴더의 사진으로 테스트하면 학습 안 한 사진으로 정확도를 확인할 수 있어요!

정확도 확인 (mAP 점수)
metrics = model.val(data='/content/dataset/dataset.yaml') print(f"\n📊 정확도 결과:") print(f" mAP50 : {metrics.box.map50:.1%}") # 기본 정확도 print(f" mAP50-95: {metrics.box.map:.1%}") # 엄격한 정확도 print(f" Precision: {metrics.box.mp:.1%}") print(f" Recall : {metrics.box.mr:.1%}")
